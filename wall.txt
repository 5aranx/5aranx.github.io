def week_rows(calendar_weeks):
    """GraphQL weeks -> [{"s": sunday-iso, "c": [7 counts or None]}]."""
    rows = []
    for wk in calendar_weeks:
        c, sunday = [None] * 7, None
        for day in wk["contributionDays"]:
            d, wd = dt.date.fromisoformat(day["date"]), day["weekday"]  # 0 = Sunday
            c[wd] = day["contributionCount"]
            sunday = d - dt.timedelta(days=wd)
        if sunday:
            rows.append({"s": sunday.isoformat(), "c": c})
    return rows
def day_list(rows):
    out = []
    for wk in rows:
        s = dt.date.fromisoformat(wk["s"])
        out += [(s + dt.timedelta(days=i), c) for i, c in enumerate(wk["c"]) if c is not None]
    return out
def month_buckets(days, today):
    keys, y, m = [], today.year, today.month
    for _ in range(12):
        keys.append((y, m))
        m -= 1
        if m == 0:
            y, m = y - 1, 12
    keys.reverse()
    totals = defaultdict(int)
    for d, c in days:
        totals[(d.year, d.month)] += c
    return [{"key": f"{y}-{m:02d}", "label": calendar.month_abbr[m], "count": totals[(y, m)]} for y, m in keys]
def streaks(days, today):
    counts = {d: c for d, c in days}
    d = today if counts.get(today, 0) > 0 else today - DAY  # today may simply not be done yet
    cur = 0
    while counts.get(d, 0) > 0:
        cur += 1
        d -= DAY
    best = run = 0
    best_end = None
    for d, c in sorted(days):
        if c > 0:
            run += 1
            if run > best:
                best, best_end = run, d
        else:
            run = 0
    return cur, best, best_end

export const HC_UNION = [
  [0, 12], [12, 0], [16, 0], [16, 32], [29, 32], [34, 27], [34, 16], [50, 0], [88, 0], [88, 16],
  [57, 16], [50, 23], [50, 57], [57, 64], [88, 64], [72, 80], [50, 80], [34, 64], [34, 47],
  [33, 48], [16, 48], [16, 80], [0, 80],
];
export function inside(x, y, polygon = HC_UNION) {
  let result = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const [xi, yi] = polygon[i];
    const [xj, yj] = polygon[j];
    if ((yi > y) !== (yj > y) && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi) result = !result;
  }
  return result;
}
export function rng(seed) {
  let state = seed >>> 0;
  return () => {
    state = (state * 1664525 + 1013904223) >>> 0;
    return state / 4294967296;
  };
}
const reducedQuery = matchMedia("(prefers-reduced-motion: reduce)");
export function prefersStill() {
  return reducedQuery.matches || new URLSearchParams(location.search).has("still");
}

  reducedQuery.addEventListener("change", () => {
    if (reducedQuery.matches) callback();

}
export async function loadJSON(url) {
  const response = await fetch(url);
  if (!response.ok) throw new Error(`${url}: ${response.status}`);
  return response.json();
}
export function fitCanvas(canvas) {
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  const width = canvas.clientWidth;
  const height = canvas.clientHeight;
  canvas.width = Math.max(1, Math.round(width * dpr));
  canvas.height = Math.max(1, Math.round(height * dpr));
  const context = canvas.getContext("2d");
  context.setTransform(dpr, 0, 0, dpr, 0, 0);
  return { context, width, height, dpr };
}
export function onWidthResize(callback) {
  let width = window.innerWidth;
  let height = window.innerHeight;
  let frame = 0;
  window.addEventListener("resize", () => {
    if (window.innerWidth === width && Math.abs(window.innerHeight - height) < 120) return;
    width = window.innerWidth;
    height = window.innerHeight;
    cancelAnimationFrame(frame);
    frame = requestAnimationFrame(callback);
  });
}
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
export function formatDay(iso) {
  const [, month, day] = iso.split("-").map(Number);
  return `${day} ${MONTHS[month - 1]}`;
}
export function isoAdd(iso, days) {
  const [year, month, day] = iso.split("-").map(Number);
  return new Date(Date.UTC(year, month - 1, day + days)).toISOString().slice(0, 10);

def page(title, desc, body):
    return f"""<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">
<meta http-equiv="Content-Security-Policy" content="default-src 'none'; style-src 'unsafe-inline'; img-src 'self' data:; base-uri 'none'; form-action 'none'">
<meta name="referrer" content="no-referrer"><title>{html.escape(title)}</title><meta name="description" content="{html.escape(desc)}"><style>{CSS}</style></head>
<body><main><nav><a href="/">home</a><a href="/site/">editor</a><a href="/blog/">blog</a></nav>{body}</main></body></html>"""
posts = []
for f in sorted((ROOT/"blog-src").glob("*.md")):
    raw = f.read_text()
    m = re.match(r"---\n(.*?)\n---\n(.*)", raw, re.S)
    fm = dict(l.split(": ", 1) for l in m.group(1).splitlines() if ": " in l)
    slug = f.stem
    out = ROOT/"blog"/slug; out.mkdir(parents=True, exist_ok=True)
    body = markdown.markdown(m.group(2), extensions=["fenced_code", "tables"])
    (out/"index.html").write_text(page(fm["title"], fm.get("description", ""), f'<h1>{html.escape(fm["title"])}</h1><p class="meta">{fm.get("date","")} · {html.escape(fm.get("tags",""))}</p>{body}'))
    posts.append((fm.get("date", ""), slug, fm))
posts.sort(reverse=True)
items = "".join(f'<li class="post"><a href="/blog/{s}/">{html.escape(fm["title"])}</a><div class="meta">{d}</div><div>{html.escape(fm.get("description",""))}</div></li>' for d, s, fm in posts)
(ROOT/"blog"/"index.html").write_text(page("Field notes", "Notes from Saransh Singh", f"<h1>Field notes</h1><ul>{items}</ul>"))
print(f"built {len(posts)} posts")
